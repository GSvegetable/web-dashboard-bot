// ✨ 核心修复：只卡黑边边界，绝不强制居中打断用户手势 function applyCanvasTransform() { const vw = window.innerWidth; const vh = window.innerHeight; // 画布原点在 [-50vw, -50vh] const limitXMin = vw * 0.5 * currentScale; // 防止左侧漏黑边 const limitXMax = vw * (1 - 1.5 * currentScale); // 防止右侧漏黑边 const limitYMin = vh * 0.5 * currentScale; const limitYMax = vh * (1 - 1.5 * currentScale); let cX = canvasOffsetX; let cY = canvasOffsetY; // 如果画面比屏幕还小，直接卡死偏移量，绝对不重置中心 if (limitXMin > limitXMax) { // 仅限制在当前合理范围，不强制居中，把控制权还给用户手指 cX = Math.min(limitXMin, Math.max(limitXMax, cX)); cY = Math.min(limitYMin, Math.max(limitYMax, cY)); } else { // 如果画面比屏幕大，严格限制拖拽边界 cX = Math.min(limitXMax, Math.max(limitXMin, cX)); cY = Math.min(limitYMax, Math.max(limitYMin, cY)); } canvasOffsetX = cX; canvasOffsetY = cY; canvasArea.style.transform = `translate(${cX}px, ${cY}px) scale(${currentScale})`; }
